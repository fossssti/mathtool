--Solve--

1.  python mathtool.py
    → справка без параметров; код 0

2.  python mathtool.py --help
    → справка; код 0

3.  python mathtool.py xyz
    → ошибка argparse (неизвестная команда); код 2

4.  python mathtool.py solve -a 2 -b 5
    → ОШИБКА: укажите все три коэффициента либо ни одного; код 1

5.  python mathtool.py solve -a abc -b 2 -c 3
    → ошибка argparse (не целое число); код 2

6.  python mathtool.py solve -a 20000 -b 5 -c 7
    → ОШИБКА: коэффициент A вне допустимого диапазона; код 1

7.  python mathtool.py solve -a 1 -b -7 -c 12
    → квадратное; D = 1; x1 = 4.000; x2 = 3.000; код 0

8.  python mathtool.py solve -a 1 -b -6 -c 9
    → квадратное; D = 0; x = 3.000; код 0

9.  python mathtool.py solve -a 1 -b 3 -c 10
    → квадратное; D = -31; действительных корней нет; код 0

10. python mathtool.py solve -a 0 -b 4 -c -8
    → линейное; x = 2.000; код 0

11. python mathtool.py solve -a 0 -b 0 -c 9
    → ОШИБКА: это не уравнение, неизвестное отсутствует; код 1

12. python mathtool.py solve -a 1 -b -4 -c 3
    → квадратное; D = 4; x1 = 3.000; x2 = 1.000; код 0

13. python mathtool.py solve -c 3 -a 1 -b -4
    → то же, что в 12 

14. python mathtool.py solve -a 10000 -b 0 -c 2
    → квадратное; D = -80000; действительных корней нет; код 0 (граница диапазона)

15. python mathtool.py solve -a 10001 -b 0 -c 2
    → ОШИБКА: коэффициент A вне допустимого диапазона; код 1

16. python mathtool.py solve -y 1 -b -4 -c 3
    → ошибка argparse (неизвестный параметр); код 2

17. python mathtool.py solve -b
    → ошибка argparse (параметр без значения); код 2

18. python mathtool.py solve   (ввести: 2, -5, 3)
    → квадратное; D = 1; x1 = 3.000; x2 = 1.000; код 0

19. python mathtool.py solve   (ввести: 2, y, 3)
    → ОШИБКА: коэффициент не является целым числом; код 1

20. python mathtool.py solve -a 2 -a 3 -b 1 -c 2
    → то же, что в 9 с A = 3; код 0

--Stats--

21. python mathtool.py stats --input numbers.txt
    → Количество 3; сумма 10.000; ср. арифм. 3.333; сумма кв. 42.000; ср. кв. 3.742;
      Дисперсия 2.889; СКО 1.700; станд. откл. 2.082; наименьшее 1.000; наибольшее 5.000;
      положительных 3; отрицательных 0; код 0

22. echo "4 1 5" | python mathtool.py stats
    → то же, что в 21 

23. echo "7" | python mathtool.py stats
    → Количество 1; дисперсия 0.000; станд. откл. НЕ СУЩЕСТВУЕТ; код 0

24. echo "" | python mathtool.py stats
    → Ошибка: последовательность пуста; код 1

25. echo "1 2 abc" | python mathtool.py stats
    → Ошибка: abc не является числом; код 1

26. python mathtool.py stats --input n.txt
    → Ошибка: [Errno 2] No such file or directory: 'n.txt'; код 1
